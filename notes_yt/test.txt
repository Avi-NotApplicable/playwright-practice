locators:
    1. page.getByRole()             ->      role (button, textbox, checkbox, etc.)
    2. page.getByText()             ->      visible text
    3. page.getByLabel()            ->      associated label text
    4. page.getByPlaceholder()      ->      placeholder text
    5. page.getByAltText()          ->      image alt text
    6. page.getByTitle()            ->      title attribute
    7. page.getByTestId()           ->      test id attribute   


other locators: page.locator('')
    1. #id                          ->      (self-explanatory)
    2. tagname                      ->      (self-explanatory)
    3. .classname                   ->      (self-explanatory)
    4. [attribute="value"]          ->      attribute
    5. #id.classnametagname         ->      multiple locators
    6. xpath="value"                ->      identify usgin its xpath
    7. :text("desiredtext")         ->      contains desired
    8. :has-text("desiredtext")     ->      container that has text
    9. :text-is("exact test")       ->      exact desired
